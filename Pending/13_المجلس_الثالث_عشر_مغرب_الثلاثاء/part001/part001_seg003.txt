من أهل الحجاج ثم قال رحمه الله قال النووي وتشبيه أمور الآخرة بأمور الدنيا إنما هو للتقريب إلى الأفهام وإلا فذرة من الآخرة خير من الأرض بأسرها وأمثالها معها أي في قوله خير لك من حمر النعم فإنما يجي على العبد من الخير في الآخرة لما له من عمل في الدنيا أعظم مما في الدنيا كلها ثم قال وفيه فضيلة من اهتدى على يديه رجل واحد للأجل العظيم الذي رتب عليه وجواز الحلف على الخبر والفتياء ولو لم يستحلف أي ولو لم يطلب منه الحلف وفي قرة عيون الموحدين قال فيه جواز الحلف حلف المفتي على ما أفتى به وزاد المصنف رحمه الله سعاله في قرة عيون الموحدين فيما يستنبط من الحديث قال وفيه الترغيب في الدعوة إلى الله وطلب الهداية لمن أراد الله هدايته ليحصل للداعي إلى الحق هذه الفضيلة العظيمة بهداية من اهتدى فلا ينبغي التفريط في المطالب العالية وبالله التوفيق وقال أيضا قبل ذلك في الصفحه السبعين بعد المئه في ذكر فائده من فوائد الحديث قال وفيه الرد على المشركين المستدلين على الشرك بكرامات الأولياء لدلالتها على فضلهم اي اذا وقعت منهم كرامه صارت عندهم علم على الفضيله ولو كان مشركا قال وامير المؤمنين علي رضي الله عنه وقع له من الكرامات ما لم يقع لغيره وله من السابقه والجهاد والفضائل ما ليس لغيره وقد خد الاخاديد واضرمها بالنار وقدف فيها من غلا فيه واعتقد فيه بعضا ما كان يعتقده هؤلاء المشركون مع اهل البيت وغيرهم فصار من اشد الصحابه رضي الله عنهم بعدا عن الشرك وشدا على من اشرك حتى حرقهم بالنار يعني في وقعته المشهوره لما امر مولاه فانظر ان قنبر ان يوقد النار فوضع فيها من وضعه ممن زعم له شيئا من الالوهيه كان عليا بريئا من تلك الدعوه وقال في ذلك لما رايت الامر امرا منكرا اجدت نالي ودعوت قنبرا فلو ادرك علي رضي الله عنه هؤلاء المعظمين له فوق ما لهم الرتبه فصنع بهم وقال اصلافهم ممن رفعه فوق مقام له ولم ينكر احد من الصحابه على علي ما فعل الا التحريق واما التقتيل فانهم اقروه عليه قال وكذلك عمرو الخطار مع ما اعطي منه الكرمات صار من ابعاد الصحابه عن الشرك وذراعه وهؤلاء افضل اهل الكرمات فما زادهم ذلك الا قوه في التوحيد وشده على اهل الشرك والتنديد اي لم يغروا بتلك الكرمات التي جرت لهم قال كما جرى لعمر رضي الله عنه في الاستسقاء بالعباس وتعمية قبل دانيال لما وجده الصحابه في بيت مال الهرمزان كما ان المعجزات انما زادت الرسل قوه في الدعوه الى التوحيد وشده على اهل الشرك والانكار عليهم وجهادهم لكن قد يقع من الاحوال الشيطانيه لمن يستحوذ عليه الشيطان فانساه ذكر ربه ما قد يرتبس على الجفهال الذين قد تلبسوا بالشرك ويظنون ان ذلك كرامات وهي مكة الشيطان واغوائه لمن لم يعرف الحق من الباطل وقد قال تعالى لنبيه محمد صلى الله عليه وسلم فاستمسك بالذي احيى اليه انك على صراط مستقيم فكذلك يجب على كل احد ان يطلب الحق من القران بتدبره فانه الصراط المستقيم ولا يلكفت الى ما زخرفته الشياطين كما اغتر به من اغتر في هذه الامه ومن قبلهم قال ابن القيم رحمه الله تعالى حذق بقلبك في النصوص كمثل ما قد حذقوا في الرأي طول زمان وترك رسوم الخلق لا تعدى فيها في استعد ما يغنيك عن دبران فاذا عنا للمرء حال من الاحوال المخالفه للشرع فانه لا ينبغي له ان يغتر به ولو كان صادرا ممن كان في رتبه عظيمه فما خالف الشرع فانه ملقا مضطرح مهما جرى لصاحبه من احوال تظن انها كرامات وهي في الحقيقه احوال قال الاخضري في الدره الثانيه قال الاخضري في نظمه في باب الرقايق والتصوف كما يقولون قال اذا رايت رجلا يطير وفوق ماء البحر قد يسير ولم يكن وفق شروط
